// api/analisar-foto.js // Function serverless da Vercel — roda no servidor, nunca no navegador. // A chave da Anthropic fica em uma variável de ambiente (ANTHROPIC_API_KEY), // configurada no painel da Vercel, e nunca aparece no código público. export default async function handler(req, res) { // Permite chamadas vindas do navegador (mesmo domínio, mas deixamos explícito) res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { return res.status(200).end(); } if (req.method !== 'POST') { return res.status(405).json({ error: 'Método não permitido, use POST.' }); } try { const { imageBase64, mediaType, prompt } = req.body; if (!imageBase64 || !mediaType) { return res.status(400).json({ error: 'Envie imageBase64 e mediaType.' }); } const apiKey = process.env.ANTHROPIC_API_KEY; if (!apiKey) { return res.status(500).json({ error: 'ANTHROPIC_API_KEY não configurada no servidor.' }); } const promptFinal = prompt || 'Olhe a foto deste prato de comida. Liste os alimentos que você identifica, estime a porção de cada um e dê uma estimativa de calorias por item e o total aproximado do prato. Também aponte macronutrientes aproximados (proteína, carboidrato, gordura) do prato todo. Responda em português, de forma direta, organizada em tópicos, sem introdução. Deixe claro que é uma estimativa visual, não uma medição exata.'; const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01', }, body: JSON.stringify({ model: 'claude-sonnet-4-6', max_tokens: 1000, messages: [{ role: 'user', content: [ { type: 'image', source: { type: 'base64', media_type: mediaType, data: imageBase64 } }, { type: 'text', text: promptFinal } ] }] }) }); const data = await response.json(); if (!response.ok) { return res.status(response.status).json({ error: data.error?.message || 'Erro na API da Anthropic.' }); } const texto = (data.content || []).map(b => b.text || '').join('\n'); return res.status(200).json({ texto }); } catch (err) { return res.status(500).json({ error: err.message }); } }
